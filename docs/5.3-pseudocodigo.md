## Algoritmo de Evaluación de Fórmulas y Pilas

```pseint
SubProceso resultado <- Evaluar(tokens)
    Definir pilaNumeros, pilaOperadores Como Caracter
    Definir i Como Entero
    Definir token, op Como Caracter
    Definir izquierdo, derecho Como Real
    
    Para i <- 1 Hasta CantidadTokens(tokens) Con Paso 1
        token <- ObtenerToken(tokens, i)
        
        Si EsNumero(token) Entonces
            Apilar(pilaNumeros, ConvertirANumero(token))
            
        SinoSi token = "(" Entonces
            Apilar(pilaOperadores, token)
            
        SinoSi token = ")" Entonces
            Mientras TopePila(pilaOperadores) <> "(" Hacer
                op <- Desapilar(pilaOperadores)
                derecho <- Desapilar(pilaNumeros)
                izquierdo <- Desapilar(pilaNumeros)
                Apilar(pilaNumeros, AplicarOperacion(op, izquierdo, derecho))
            FinMientras
            Desapilar(pilaOperadores)
            
        SinoSi EsOperador(token) Entonces
            Mientras PilaNoVacia(pilaOperadores) Y Precedencia(TopePila(pilaOperadores)) >= Precedencia(token) Hacer
                op <- Desapilar(pilaOperadores)
                derecho <- Desapilar(pilaNumeros)
                izquierdo <- Desapilar(pilaNumeros)
                Apilar(pilaNumeros, AplicarOperacion(op, izquierdo, derecho))
            FinMientras
            Apilar(pilaOperadores, token)
            
        Sino
            Si CeldaVacia(token) Entonces
                Escribir "Error: #REF!"
            Sino
                Apilar(pilaNumeros, ResolverCelda(token))
            FinSi
        FinSi
    FinPara
    
    Mientras PilaNoVacia(pilaOperadores) Hacer
        op <- Desapilar(pilaOperadores)
        derecho <- Desapilar(pilaNumeros)
        izquierdo <- Desapilar(pilaNumeros)
        Apilar(pilaNumeros, AplicarOperacion(op, izquierdo, derecho))
    FinMientras
    
    resultado <- Desapilar(pilaNumeros)
FinSubProceso
Proceso RecalcularTodo
    Definir TotalFilas, TotalColumnas, f, c Como Entero
    Definir idCelda, valorGuardado Como Caracter
    
    TotalFilas <- 15
    TotalColumnas <- 10
    
    Para f <- 1 Hasta TotalFilas Con Paso 1
        Para c <- 0 Hasta TotalColumnas - 1 Con Paso 1
            idCelda <- ConstruirIdCelda(c, f)
            valorGuardado <- ObtenerValorGuardado(idCelda)
            
            Si valorGuardado <> "" Entonces
                MostrarEnPantalla(idCelda, ProcesarContenidoCelda(valorGuardado))
            Sino
                MostrarEnPantalla(idCelda, "")
            FinSi
        FinPara
    FinPara
FinProceso
